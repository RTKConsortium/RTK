# Doxygen documentation for RTK.
#
# The configuration is expressed as DOXYGEN_* variables in DoxygenConfig.cmake
# and consumed by CMake's doxygen_add_docs() (from the FindDoxygen module)
# instead of the hand-maintained doxygen.config.in Doxyfile used previously.
# This mirrors the approach used by ITK since ITK 5.3.
#   https://cmake.org/cmake/help/latest/module/FindDoxygen.html

# Only Doxygen and Perl are actually used; Gnuplot/HTMLHelp/Wget are no
# longer needed since the ITK tag file is downloaded with file(DOWNLOAD).
find_package(Perl)
find_package(Doxygen)

if(NOT DOXYGEN_FOUND)
  message(
    STATUS
    "Doxygen not found - the Documentation target will not be available."
  )
  return()
endif()

# Output directory for the generated documentation. The published pages live at
# https://www.openrtk.org/Doxygen/, whose html/ folder is copied from here.
set(RTK_DOXYGEN_OUTPUT_DIR "${PROJECT_BINARY_DIR}/Doxygen")

# Information on how to retrieve the ITK documentation tag file
set(
  ITK_DOXYGEN_TAG_LOCATION
  "https://itk.org/files/NightlyDoxygen/InsightDoxygenDocTag.gz"
)
set(
  ITK_DOXYGEN_COMPRESSED_TAG_FILE
  ${RTK_DOXYGEN_OUTPUT_DIR}/InsightDoxygen.tag.gz
)
set(ITK_DOXYGEN_TAG_FILE ${RTK_DOXYGEN_OUTPUT_DIR}/InsightDoxygen.tag)

# Get the ITK documentation tag file so that references to itk:: classes are
# cross-linked to the ITK documentation.
if(NOT EXISTS ${ITK_DOXYGEN_COMPRESSED_TAG_FILE})
  file(
    DOWNLOAD
      ${ITK_DOXYGEN_TAG_LOCATION}
      ${ITK_DOXYGEN_COMPRESSED_TAG_FILE}
    TIMEOUT 60
    STATUS statusITKDoxygenTagFile
    SHOW_PROGRESS
  )
  list(GET statusITKDoxygenTagFile 0 statusITKDoxygenTagFile)
  if(statusITKDoxygenTagFile)
    file(REMOVE ${ITK_DOXYGEN_COMPRESSED_TAG_FILE})
  endif()
endif()
if(EXISTS ${ITK_DOXYGEN_COMPRESSED_TAG_FILE})
  find_program(GZIP_TOOL NAMES gzip)
  if(GZIP_TOOL)
    execute_process(
      COMMAND
        ${GZIP_TOOL} -dkf ${ITK_DOXYGEN_COMPRESSED_TAG_FILE}
    )
    set(
      DOXYGEN_TAGFILES
      "${ITK_DOXYGEN_TAG_FILE}=https://www.itk.org/Doxygen/html/"
    )
  endif()
else()
  set(DOXYGEN_TAGFILES "")
endif()

# The input filter rewrites the comments of the vendored vxl/vnl sources and
# groups RTK classes into Doxygen sub-groups.
if(PERL_FOUND)
  set(
    RTK_DOXYGEN_INPUT_FILTER
    "${PERL_EXECUTABLE} ${RTK_DOXYGEN_OUTPUT_DIR}/itkdoxygen.pl"
  )
  configure_file(
    "${PROJECT_SOURCE_DIR}/documentation/Doxygen/itkdoxygen.pl.in"
    "${RTK_DOXYGEN_OUTPUT_DIR}/itkdoxygen.pl"
    @ONLY
  )
else()
  set(RTK_DOXYGEN_INPUT_FILTER)
endif()

# ---------------------------------------------------------------------------
# Output formats. HTML is generated by default, the others are opt-in.
# ---------------------------------------------------------------------------
include(CMakeDependentOption)

cmake_dependent_option(
  RTK_DOXYGEN_HTML
  "Doxygen will generate HTML output"
  ON
  "RTK_BUILD_DOXYGEN"
  OFF
)
cmake_dependent_option(
  RTK_DOXYGEN_DOCSET
  "Doxygen will generate additional index files for Xcode"
  OFF
  "RTK_BUILD_DOXYGEN"
  OFF
)
cmake_dependent_option(
  RTK_DOXYGEN_CHM
  "Doxygen will generate Microsoft HTML help"
  OFF
  "RTK_BUILD_DOXYGEN"
  OFF
)
cmake_dependent_option(
  RTK_DOXYGEN_QHP
  "Doxygen will generate Qt Compressed Help"
  OFF
  "RTK_BUILD_DOXYGEN"
  OFF
)
cmake_dependent_option(
  RTK_DOXYGEN_ECLIPSEHELP
  "Doxygen will generate Eclipse help plugin"
  OFF
  "RTK_BUILD_DOXYGEN"
  OFF
)
cmake_dependent_option(
  RTK_DOXYGEN_LATEX
  "Doxygen will generate LaTeX output"
  OFF
  "RTK_BUILD_DOXYGEN"
  OFF
)
cmake_dependent_option(
  RTK_DOXYGEN_RTF
  "Doxygen will generate RTF output optimized for Word 97"
  OFF
  "RTK_BUILD_DOXYGEN"
  OFF
)
cmake_dependent_option(
  RTK_DOXYGEN_XML
  "Doxygen will generate XML output"
  OFF
  "RTK_BUILD_DOXYGEN"
  OFF
)
cmake_dependent_option(
  RTK_DOXYGEN_SERVER_BASED_SEARCH
  "Doxygen HTML will use PHP server based search."
  ON
  RTK_DOXYGEN_HTML
  OFF
)
cmake_dependent_option(
  RTK_DOXYGEN_WARNINGS_ONLY
  "Doxygen emits warnings as errors without generating output or implicit graphs"
  OFF
  "RTK_BUILD_DOXYGEN"
  OFF
)

mark_as_advanced(
  RTK_DOXYGEN_HTML
  RTK_DOXYGEN_DOCSET
  RTK_DOXYGEN_CHM
  RTK_DOXYGEN_QHP
  RTK_DOXYGEN_ECLIPSEHELP
  RTK_DOXYGEN_LATEX
  RTK_DOXYGEN_RTF
  RTK_DOXYGEN_XML
  RTK_DOXYGEN_SERVER_BASED_SEARCH
  RTK_DOXYGEN_WARNINGS_ONLY
)

# ---------------------------------------------------------------------------
# Inputs and target
# ---------------------------------------------------------------------------
set(
  RTK_DOXYGEN_INPUT
  "${PROJECT_SOURCE_DIR}/include"
  "${PROJECT_SOURCE_DIR}/documentation/Doxygen"
)
list(REMOVE_DUPLICATES RTK_DOXYGEN_INPUT)

# DoxygenConfig.cmake is a replacement for configuring a doxygen.config.in file.
include("${CMAKE_CURRENT_LIST_DIR}/DoxygenConfig.cmake")

if(RTK_DOXYGEN_WARNINGS_ONLY)
  foreach(
    _format
    IN
    ITEMS
      HTML
      DOCSET
      CHM
      QHP
      ECLIPSEHELP
      LATEX
      RTF
      XML
  )
    if(_format STREQUAL "CHM")
      set(DOXYGEN_GENERATE_HTMLHELP "NO")
    else()
      set(DOXYGEN_GENERATE_${_format} "NO")
    endif()
  endforeach()
  # Explicit \dot commands still render; only the implicit graph classes are
  # disabled.
  foreach(
    _graph
    IN
    ITEMS
      CLASS_GRAPH
      COLLABORATION_GRAPH
      INCLUDE_GRAPH
      INCLUDED_BY_GRAPH
      CALL_GRAPH
      CALLER_GRAPH
      GRAPHICAL_HIERARCHY
      DIRECTORY_GRAPH
  )
    set(DOXYGEN_${_graph} "NO")
  endforeach()
  set(DOXYGEN_WARN_AS_ERROR "FAIL_ON_WARNINGS")
  set(DOXYGEN_WARN_LOGFILE "${RTK_DOXYGEN_OUTPUT_DIR}/doxygen-warnings.log")
  # GENERATE_TAGFILE stays: doxygen requires at least one output and the tag
  # file is the cheapest.
endif()

doxygen_add_docs(
  Documentation
  ${RTK_DOXYGEN_INPUT}
  # -- Only build if the 'Documentation' target is requested
  WORKING_DIRECTORY "${RTK_DOXYGEN_OUTPUT_DIR}"
  # -- Some of RTK_DOXYGEN_INPUT are directories; USE_STAMP_FILE would require
  # -- only files to be listed.
  COMMENT "-- Building RTK Doxygen."
)
unset(RTK_DOXYGEN_INPUT)

message(
  STATUS
  "To generate Doxygen documentation, build the Documentation target."
)
